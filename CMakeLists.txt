# One catalogue supplies the GUI and the offline capture runner.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS study.json)
foreach(key title kind parameter_label candidate)
    string(JSON ${key} GET "${study}" ${key})
endforeach()
function(cpp_array key quote_items)
    string(JSON length LENGTH "${study}" ${key})
    math(EXPR last "${length}-1")
    set(items "")
    foreach(i RANGE 0 ${last})
        string(JSON item GET "${study}" ${key} ${i})
        if(quote_items)
            list(APPEND items "\"${item}\"")
        else()
            list(APPEND items "${item}")
        endif()
    endforeach()
    list(JOIN items ", " joined)
    set(${key} "${joined}" PARENT_SCOPE)
endfunction()
foreach(key method_labels scene_labels views methods scenes)
    cpp_array(${key} TRUE)
endforeach()
foreach(key budgets parameters)
    cpp_array(${key} FALSE)
endforeach()
