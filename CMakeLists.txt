cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/study.json" study)
string(JSON project_id GET "${study}" id)
project(${project_id} VERSION 0.1.0 LANGUAGES C CXX)

set(GFX_RESEARCH_BASE_LOCAL_PATH "" CACHE PATH "Optional local gfx-research-base v0.3 checkout")
set(GFX_RESEARCH_BASE_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
if(GFX_RESEARCH_BASE_LOCAL_PATH)
    add_subdirectory("${GFX_RESEARCH_BASE_LOCAL_PATH}" "${CMAKE_BINARY_DIR}/gfx-research-base")
else()
    include(FetchContent)
    FetchContent_Declare(research_base
        GIT_REPOSITORY https://github.com/Terrasil/gfx-research-base.git
        GIT_TAG v0.3
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(research_base)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)

# One catalogue supplies the GUI and the offline capture runner.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS study.json)
foreach(key title kind parameter_label candidate)
    string(JSON ${key} GET "${study}" ${key})
endforeach()
function(cpp_array key quote_items)
    string(JSON length LENGTH "${study}" ${key})
    math(EXPR last "${length}-1")
    set(items "")
    foreach(i RANGE 0 ${last})
        string(JSON item GET "${study}" ${key} ${i})
        if(quote_items)
            list(APPEND items "\"${item}\"")
        else()
            list(APPEND items "${item}")
        endif()
    endforeach()
    list(JOIN items ", " joined)
    set(${key} "${joined}" PARENT_SCOPE)
endfunction()
foreach(key method_labels scene_labels views methods scenes)
    cpp_array(${key} TRUE)
endforeach()
foreach(key budgets parameters)
    cpp_array(${key} FALSE)
endforeach()
configure_file(src/catalog.hpp.in generated/catalog.hpp @ONLY)
add_executable(${PROJECT_NAME} src/main.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_20)
target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(${PROJECT_NAME} PRIVATE gfx::research_base)
gfx_research_configure_executable(${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
    add_test(NAME study-plan COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/study.py" plan --profile smoke)
endif()
